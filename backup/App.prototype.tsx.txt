function PrimaryButton({
  children,
  onClick,
  loading = false,
  outline = false,
  disabled = false,
}: {
  children: React.ReactNode;
  onClick?: () => void;
  loading?: boolean;
  outline?: boolean;
  disabled?: boolean;
}) {
  return (
    <button
      disabled={disabled || loading}
      className={`primary-button ${outline ? "outline" : ""} ${loading ? "loading" : ""}`}
      onClick={onClick}
    >
      {loading ? <span className="spinner" /> : children}
    </button>
  );
}

function Stars({ count = 5, interactive = false }: { count?: number; interactive?: boolean }) {
  const [rating, setRating] = useState(count);
  return (
    <div className={`stars ${interactive ? "interactive" : ""}`}>
      {[1, 2, 3, 4, 5].map((n) => (
        <button
          key={n}
          disabled={!interactive}
          onClick={() => setRating(n)}
          aria-label={`${n} stars`}
        >
          <Icon name="star" size={22} filled={n <= rating} />
        </button>
      ))}
    </div>
  );
}

function BottomNav({
  active = "Home",
  workerMode = false,
  go,
}: {
  active?: string;
  workerMode?: boolean;
  go: (id: ScreenId) => void;
}) {
  const items = workerMode
    ? [
        ["Home", "home", "worker-dashboard"],
        ["Jobs", "briefcase", "find-work"],
        ["Earnings", "wallet", "earnings"],
        ["Profile", "user", "worker-profile"],
      ]
    : [
        ["Home", "home", "customer-dashboard"],
        ["Search", "search", "search"],
        ["Bookings", "calendar", "bookings"],
        ["Profile", "user", "customer-profile"],
      ];
  return (
    <nav className="bottom-nav">
      {items.map(([label, icon, target]) => (
        <button
          key={label}
          className={active === label ? "active" : ""}
          onClick={() => go(target as ScreenId)}
        >
          <Icon name={icon as IconName} size={21} />
          <span>{label}</span>
        </button>
      ))}
    </nav>
  );
}

function TopBar({
  title,
  back,
  action,
}: {
  title?: string;
  back?: () => void;
  action?: React.ReactNode;
}) {
  return (
    <header className="topbar">
      {back ? (
        <button className="icon-button" onClick={back}>
          <Icon name="arrow" size={22} />
        </button>
      ) : (
        <Logo compact />
      )}
      <strong>{title}</strong>
      <div>{action}</div>
    </header>
  );
}

export default function App() {
  const [current, setCurrent] = useState<ScreenId>("splash");
  const [history, setHistory] = useState<ScreenId[]>([]);
  const [direction, setDirection] = useState<"forward" | "back" | "down">("forward");
  const [role, setRole] = useState<Role | null>(null);
  const [visitedRoles, setVisitedRoles] = useState<Role[]>([]);
  const navigate = (next: ScreenId) => {
    setDirection("forward");
    setHistory((previous) => (next === "splash" ? [] : [...previous, current]));
    setCurrent(next);
  };
  const openRoleSelection = () => {
    setDirection("down");
    setHistory((previous) => [...previous, current]);
    setCurrent("home");
  };
  const selectRole = (nextRole: Role) => {
    const repeatVisit = visitedRoles.includes(nextRole);
    setRole(nextRole);
    setVisitedRoles((roles) => (roles.includes(nextRole) ? roles : [...roles, nextRole]));
    setHistory([]);
    setDirection("forward");
    setCurrent(
      repeatVisit
        ? nextRole === "customer"
          ? "customer-dashboard"
          : "worker-dashboard"
        : nextRole === "customer"
          ? "customer-welcome"
          : "worker-welcome",
    );
  };
  const goBack = () => {
    const previous = history.at(-1);
    if (!previous) return;
    setDirection("back");
    setHistory((items) => items.slice(0, -1));
    setCurrent(previous);
  };
  return (
    <main className="app-viewport">
      <div className={`screen-transition ${direction}`} key={current}>
        <PhoneScreen
          id={current}
          go={navigate}
          onBack={goBack}
          role={role}
          selectRole={selectRole}
          openRoleSelection={openRoleSelection}
          prototype
        />
      </div>
    </main>
  );
}
